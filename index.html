Instalar no Celular

Para deixar o Escutatória 60+ com ícone direto na tela inicial do celular:

  • Android (Chrome): Toque nos 3 pontinhos (⋮) no canto superior e escolha "Adicionar à tela inicial" ou "Instalar aplicativo".
  • iPhone (Safari): Toque no ícone de Compartilhar (□↑) na barra inferior e escolha "Adicionar à Tela de Início".

Conectar ao Google Planilhas

Para manter os psicólogos e o histórico salvos em uma planilha do Google sem custo:

  1. Abra uma nova planilha no Google Planilhas e vá em Extensões > Apps Script.
  2. Apague o conteúdo atual e cole o script abaixo. Depois clique em Implantar > Nova implantação escolhendo App da Web, com acesso definido para Qualquer pessoa.
  3. Copie a URL do App da Web gerada (termina com /exec), cole no campo Base via Google Planilha no Admin e clique em Salvar e Sincronizar.
function doGet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const stateSheet = ss.getSheetByName('EstadoJSON') || ss.insertSheet('EstadoJSON');
  const raw = stateSheet.getRange('A1').getValue();
  return ContentService.createTextOutput(raw || '{}')
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const bodyText = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
  const data = JSON.parse(bodyText);

  // 1. Salva snapshot completo para sincronizar o app
  const stateSheet = ss.getSheetByName('EstadoJSON') || ss.insertSheet('EstadoJSON');
  stateSheet.getRange('A1').setValue(bodyText);

  // 2. Atualiza aba legível de Psicólogos
  const psychSheet = ss.getSheetByName('Psicologos') || ss.insertSheet('Psicologos');
  psychSheet.clearContents();
  psychSheet.appendRow(['ID', 'Nome', 'CRP', 'Foco', 'Turno', 'Status', 'Atualizado Em']);
  const psychs = Array.isArray(data.psychologists) ? data.psychologists : [];
  psychs.forEach(p => {
    psychSheet.appendRow([
      p.id || '',
      p.name || '',
      p.crp || '',
      p.specialty || '',
      p.shift || '',
      p.online ? 'ONLINE' : 'OFFLINE',
      new Date().toLocaleString('pt-BR')
    ]);
  });

  // 3. Atualiza aba legível de Histórico de Atendimentos
  const histSheet = ss.getSheetByName('Historico') || ss.insertSheet('Historico');
  histSheet.clearContents();
  histSheet.appendRow(['ID', 'Idoso(a)', 'Psicologo(a)', 'Data/Hora', 'Duracao', 'Observacoes']);
  const hist = Array.isArray(data.history) ? data.history : [];
  hist.forEach(h => {
    histSheet.appendRow([
      h.id || '',
      h.seniorName || '',
      h.psychName || '',
      h.dateStr || '',
      h.duration || '',
      h.notes || ''
    ]);
  });

  return ContentService.createTextOutput(JSON.stringify({ ok: true, count: psychs.length }))
    .setMimeType(ContentService.MimeType.JSON);
}
